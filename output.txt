# Personal Tracker (REACT/DJANGO)

## Software Architecture

- Backend: Django + Django REST Framework (DRF) + PostgreSQL. Use Django Channels for WebSockets (realtime updates).
- Frontend: React (TypeScript recommended) with React Router, React Query (data fetching/caching), and a state manager (Zustand or Redux Toolkit).
- Calendar UI: FullCalendar React (excellent event UI). For todos use custom lists/boards.
- Auth: JWT (SimpleJWT) or session-based cookies. Use refresh tokens or rotating cookies for security.
- File storage: S3 (or DigitalOcean Spaces) for attachments.
- Deployment: Docker containers. Host backend on Render/Heroku/AWS ECS or Fly.io. Frontend on Vercel/Netlify.
- CI/CD: GitHub Actions for tests and deploy.

## Core Feature

- Multi-user auth (register/login)
- Personal calendars + shared calendars (per-user and group)
- Todos/tasks (assignable to users, with deadlines, priorities, attachments)
- Events with start/end, recurrence rules, reminders/notifications
- Sharing & permissions (owner/editor/viewer)
- Realtime updates (WebSocket) for edits, comments, presence
- Timezone support and daylight-savings awareness
- Offline support / local cache + sync (frontend with React Query + background sync)
- Search, filters, and basic analytics (e.g., tasks completed)

## Nonfunctional Needs

- Secure (auth, authorization, rate-limiting)
- Scalable (Postgres + caching with Redis)
- Test coverage (unit + integration)
- Accessible UI (a11y)
- Multi-tenant considerations if offering SaaS (optional)

// src/api/api.js
import axios from "axios";

const API_BASE = import.meta.env.VITE_API_BASE_URL || "http://localhost:8000/api";

let store = {
  access: null,
  refresh: null,
  onLogout: () => {},
};

export const setAuthTokens = (access, refresh) => {
  store.access = access;
  store.refresh = refresh;
};

export const setLogoutHandler = (logoutFn) => {
  store.onLogout = logoutFn;
};

const api = axios.create({
  baseURL: API_BASE,
  headers: { "Content-Type": "application/json" },
});

api.interceptors.request.use((config) => {
  if (store.access) config.headers.Authorization = `Bearer ${store.access}`;
  return config;
});

api.interceptors.response.use(
  (response) => response,
  async (error) => {
    const original = error.config;

    // if access expired, try refreshing
    if (error.response?.status === 401 && !original._retry && store.refresh) {
      original._retry = true;
      try {
        const res = await axios.post(`${API_BASE}/token/refresh/`, { refresh: store.refresh });
        store.access = res.data.access;
        localStorage.setItem("access", store.access);
        original.headers.Authorization = `Bearer ${store.access}`;
        return api(original);
      } catch (err) {
        console.warn("Token refresh failed, logging out");
        store.onLogout();
      }
    }
    return Promise.reject(error);
  }
);

export const login = async (username, password) => {
  const res = await api.post("/token/", { username, password });
  const { access, refresh } = res.data;
  setAuthTokens(access, refresh);
  localStorage.setItem("access", access);
  localStorage.setItem("refresh", refresh);
  return res.data;
};

export const logout = () => {
  store.access = null;
  store.refresh = null;
  localStorage.removeItem("access");
  localStorage.removeItem("refresh");
  store.onLogout();
};

export const ping = async () => {
  const res = await api.get("/ping/");
  return res.data;
};

export const getMe = async () => {
  try {
    const res = await api.get("/me/");
    return res.data; // { id, username, email }
  } catch (err) {
    console.warn("Failed to fetch user info");
    return null;
  }
};

// src/context/AuthProvider.jsx
import React, { createContext, useState, useEffect } from "react";
import { setAuthTokens, setLogoutHandler, login as apiLogin, getMe } from "../api/api";

export const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  // Logout function
  const logout = () => {
    setUser(null);
    localStorage.removeItem("access");
    localStorage.removeItem("refresh");
    window.location.href = "/login";
  };

  // Load tokens from localStorage and set logout handler
  const loadTokens = () => {
    const access = localStorage.getItem("access");
    const refresh = localStorage.getItem("refresh");
    if (access && refresh) {
      setAuthTokens(access, refresh);
      return true;
    }
    return false;
  };

  // Fetch current user if tokens exist
  const fetchUser = async () => {
    if (!loadTokens()) {
      setLoading(false);
      return;
    }
    setLogoutHandler(logout);
    try {
      const me = await getMe();
      setUser(me);
    } catch (err) {
      console.warn("Failed to fetch user info:", err);
      logout();
    } finally {
      setLoading(false);
    }
  };

  // Login function
  const login = async (username, password) => {
    setLoading(true);
    try {
      await apiLogin(username, password);
      const me = await getMe();
      setUser(me);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchUser();
  }, []);

  const value = {
    user,
    loading,
    login,
    logout,
    setUser, // optional if you want to allow manual updates
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
};

// src/pages/LoginPage.jsx
import React, { useState } from "react";
import { login } from "../api/api";
import { useNavigate } from "react-router-dom";

export default function LoginPage() {
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const navigate = useNavigate();

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    try {
      await login(username, password);
      navigate("/dashboard");
    } catch {
      setError("Invalid credentials");
    }
  };

  return (
    <div className="login-page">
      <form onSubmit={handleSubmit}>
        <h2>Login</h2>
        <input
          type="text"
          placeholder="Username"
          value={username}
          onChange={(e) => setUsername(e.target.value)}
        />
        <input
          type="password"
          placeholder="Password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
        />
        <button type="submit">Login</button>
        {error && <p style={{ color: "red" }}>{error}</p>}
      </form>
    </div>
  );
}

    const refresh = localStorage.getItem("refresh");
    if (access && refresh) {
      setAuthTokens(access, refresh);
      return true;
    }
    return false;
  };

  // Fetch current user if tokens exist
  const fetchUser = async () => {
    if (!loadTokens()) {
      setLoading(false);
      return;
    }
    setLogoutHandler(logout);
    try {
      const me = await getMe();
      setUser(me);
    } catch (err) {
      console.warn("Failed to fetch user info:", err);
      logout();
    } finally {
      setLoading(false);
    }
  };

  // Login function
  const login = async (username, password) => {
    setLoading(true);
    try {
      await apiLogin(username, password);
      const me = await getMe();
      setUser(me);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchUser();
  }, []);

  const value = {
    user,
    loading,
    login,
    logout,
    setUser, // optional if you want to allow manual updates
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
};

import React, { useState } from "react";
import { login } from "../api/api";
import { useNavigate } from "react-router-dom";

export default function LoginPage() {
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const navigate = useNavigate();

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    try {
      await login(username, password);
      navigate("/dashboard");
    } catch {
      setError("Invalid credentials");
    }
  };

  return (
    <div className="login-page">
      <form onSubmit={handleSubmit}>
        <h2>Login</h2>
        <input
          type="text"
          placeholder="Username"
          value={username}
          onChange={(e) => setUsername(e.target.value)}
        />
        <input
          type="password"
          placeholder="Password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
        />
        <button type="submit">Login</button>
        {error && <p style={{ color: "red" }}>{error}</p>}
      </form>
    </div>
  );
}
